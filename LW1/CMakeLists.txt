cmake_minimum_required(VERSION 3.20)
project(CG_Lab1_Octahedron LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
find_package(glfw3 REQUIRED)
find_package(glm REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(IMGUI REQUIRED IMPORTED_TARGET imgui)
find_path(IMGUI_BACKENDS_DIR imgui_impl_vulkan.cpp
  HINTS ${IMGUI_BACKENDS_DIR} /usr/share/doc/libimgui-dev/examples/backends)
if(NOT IMGUI_BACKENDS_DIR)
  message(FATAL_ERROR "Set IMGUI_BACKENDS_DIR to the ImGui backends source directory")
endif()
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
foreach(stage vert frag)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/octahedron.${stage}.spv
    COMMAND ${GLSLANG_VALIDATOR} -V ${CMAKE_CURRENT_SOURCE_DIR}/shaders/octahedron.${stage} -o ${CMAKE_CURRENT_BINARY_DIR}/octahedron.${stage}.spv
    DEPENDS shaders/octahedron.${stage} VERBATIM)
endforeach()
add_custom_target(shaders DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/octahedron.vert.spv ${CMAKE_CURRENT_BINARY_DIR}/octahedron.frag.spv)
add_executable(octahedron src/main.cpp ${IMGUI_BACKENDS_DIR}/imgui_impl_glfw.cpp ${IMGUI_BACKENDS_DIR}/imgui_impl_vulkan.cpp)
add_dependencies(octahedron shaders)
target_link_libraries(octahedron PRIVATE Vulkan::Vulkan glfw glm::glm PkgConfig::IMGUI)
find_path(IMGUI_BACKENDS_INCLUDE_DIR imgui_impl_vulkan.h HINTS ${IMGUI_INCLUDE_DIRS} PATH_SUFFIXES backends REQUIRED)
target_include_directories(octahedron PRIVATE ${IMGUI_BACKENDS_DIR} ${IMGUI_BACKENDS_INCLUDE_DIR})
target_compile_definitions(octahedron PRIVATE GLFW_INCLUDE_NONE SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(octahedron PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
include(CTest)
add_executable(geometry_test tests/geometry.cpp)
target_link_libraries(geometry_test PRIVATE glm::glm)
add_test(NAME geometry COMMAND geometry_test)
